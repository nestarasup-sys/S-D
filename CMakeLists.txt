cmake_minimum_required(VERSION 3.24)
project(StudioDark VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.8)
enable_testing()
add_subdirectory(modules/core)
add_subdirectory(modules/storage)
add_subdirectory(modules/composition)
add_subdirectory(apps/render-worker)
add_subdirectory(apps/desktop)
add_subdirectory(tests)
